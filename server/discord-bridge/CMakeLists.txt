cmake_minimum_required(VERSION 3.22)
project(nami_discord_bridge LANGUAGES CXX)
# Use the official SDK downloaded by the administrator from the Developer Portal.
set(DISCORD_SDK_INCLUDE_DIR "" CACHE PATH "Directory containing discordpp.h and cdiscord.h")
set(DISCORD_SDK_LIBRARY "" CACHE FILEPATH "Official SDK .so, .dylib or Windows import .lib")
if(NOT EXISTS "${DISCORD_SDK_INCLUDE_DIR}/discordpp.h" OR NOT EXISTS "${DISCORD_SDK_INCLUDE_DIR}/cdiscord.h" OR NOT EXISTS "${DISCORD_SDK_LIBRARY}")
  message(FATAL_ERROR "Set DISCORD_SDK_INCLUDE_DIR and DISCORD_SDK_LIBRARY to the official Social SDK")
endif()
find_package(Threads REQUIRED)
add_executable(nami-discord-bridge main.cpp)
target_compile_features(nami-discord-bridge PRIVATE cxx_std_20)
target_include_directories(nami-discord-bridge PRIVATE "${DISCORD_SDK_INCLUDE_DIR}")
target_link_libraries(nami-discord-bridge PRIVATE "${DISCORD_SDK_LIBRARY}" Threads::Threads)
get_filename_component(SDK_LIBRARY_DIR "${DISCORD_SDK_LIBRARY}" DIRECTORY)
if(APPLE)
  set(BRIDGE_LIBRARY_PATH "@loader_path")
else()
  set(BRIDGE_LIBRARY_PATH "$ORIGIN")
endif()
set_target_properties(nami-discord-bridge PROPERTIES BUILD_RPATH "${SDK_LIBRARY_DIR};${BRIDGE_LIBRARY_PATH}")
